## 🚀 一键启动脚本

### 创建启动脚本
```powershell
# 创建一键启动脚本
New-Item -ItemType File -Name "启动Memorin项目.ps1" -Path "."

# 编辑脚本内容
@"
#!/usr/bin/env pwsh

Write-Host "🚀 Memorin项目一键启动脚本" -ForegroundColor Cyan
Write-Host "=" * 50

# 检查环境
Write-Host "`n📋 第一步：环境检查"
if (-not (Get-Command node -ErrorAction SilentlyContinue)) {
    Write-Host "❌ Node.js未安装" -ForegroundColor Red
    exit 1
}

if (-not (Get-Command java -ErrorAction SilentlyContinue)) {
    Write-Host "❌ Java未安装" -ForegroundColor Red
    exit 1
}

if (-not (Get-Command mvn -ErrorAction SilentlyContinue)) {
    Write-Host "❌ Maven未安装" -ForegroundColor Red
    exit 1
}

Write-Host "✅ 环境检查通过" -ForegroundColor Green

# 启动Nacos
Write-Host "`n🔧 第二步：启动Nacos服务器"
if (Test-Path "nacos-server") {
    Set-Location "nacos-server"
    Start-Process -NoNewWindow powershell -ArgumentList "-Command", ".\bin\startup.cmd"
    Write-Host "✅ Nacos启动命令已执行" -ForegroundColor Green
    Set-Location ".."
} else {
    Write-Host "⚠️  Nacos目录不存在，请手动启动" -ForegroundColor Yellow
}

# 等待Nacos启动
Write-Host "⏳ 等待Nacos启动 (10秒)..."
Start-Sleep -Seconds 10

# 启动后端服务
Write-Host "`n🔧 第三步：启动后端微服务"
$services = @(
    @{Name="用户服务"; Path="Memorin-rebuild\backend\memorin-user-service"; Port=8081},
    @{Name="知识库服务"; Path="Memorin-rebuild\backend\memorin-knowledge-service"; Port=8082},
    @{Name="复习服务"; Path="Memorin-rebuild\backend\memorin-review-service"; Port=8083},
    @{Name="统计服务"; Path="Memorin-rebuild\backend\memorin-statistics-service"; Port=8084},
    @{Name="API网关"; Path="Memorin-rebuild\backend\memorin-gateway"; Port=8080}
)

foreach ($service in $services) {
    Write-Host "启动 $($service.Name)..."
    if (Test-Path $service.Path) {
        Start-Process -NoNewWindow powershell -ArgumentList "-Command", "Set-Location '$($service.Path)'; mvn spring-boot:run '-Dspring-boot.run.profiles=dev'"
        Write-Host "✅ $($service.Name)启动命令已执行" -ForegroundColor Green
    } else {
        Write-Host "❌ $($service.Path) 不存在" -ForegroundColor Red
    }
}

# 启动前端
Write-Host "`n🔧 第四步：启动前端开发环境"
if (Test-Path "Memorin-rebuild\frontend") {
    Set-Location "Memorin-rebuild\frontend"
    
    # 检查是否已安装依赖
    if (-not (Test-Path "node_modules")) {
        Write-Host "📦 安装前端依赖..."
        npm install
    }
    
    Start-Process -NoNewWindow powershell -ArgumentList "-Command", "npm run dev"
    Write-Host "✅ 前端启动命令已执行" -ForegroundColor Green
    Set-Location "..\.."
} else {
    Write-Host "❌ 前端目录不存在" -ForegroundColor Red
}

Write-Host "`n" + "=" * 50
Write-Host "🎉 启动命令已全部执行！" -ForegroundColor Green
Write-Host ""
Write-Host "📊 服务状态检查："
Write-Host "   Nacos: http://localhost:8848/nacos"
Write-Host "   API网关: http://localhost:8080"
Write-Host "   前端: http://localhost:5173"
Write-Host "   Swagger文档: http://localhost:8080/swagger-ui/index.html"
Write-Host ""
Write-Host "⏳ 请等待服务完全启动 (约2-3分钟)"
Write-Host "🔍 可以使用 'netstat -ano | findstr :8080' 检查端口占用"
"@ | Out-File -FilePath "启动Memorin项目.ps1" -Encoding UTF8
```

### 运行启动脚本
```powershell
# 运行一键启动脚本
.\启动Memorin项目.ps1
```

## 🔍 服务状态验证

### 1. 健康检查脚本
```powershell
# 创建健康检查脚本
@"
Write-Host "🔍 Memorin服务健康检查" -ForegroundColor Cyan
Write-Host "=" * 40

$services = @(
    @{Name="Nacos"; Url="http://localhost:8848/nacos"},
    @{Name="API网关"; Url="http://localhost:8080/actuator/health"},
    @{Name="用户服务"; Url="http://localhost:8081/actuator/health"},
    @{Name="知识库服务"; Url="http://localhost:8082/actuator/health"},
    @{Name="复习服务"; Url="http://localhost:8083/actuator/health"},
    @{Name="统计服务"; Url="http://localhost:8084/actuator/health"}
)

foreach ($service in $services) {
    try {
        $response = Invoke-RestMethod -Uri $service.Url -Method Get -TimeoutSec 5
        Write-Host "✅ $($service.Name): 正常" -ForegroundColor Green
    } catch {
        Write-Host "❌ $($service.Name): 异常" -ForegroundColor Red
    }
}

Write-Host "`n📊 前端开发环境:"
try {
    $response = Invoke-RestMethod -Uri "http://localhost:5173" -Method Get -TimeoutSec 5
    Write-Host "✅ 前端服务: 正常" -ForegroundColor Green
} catch {
    Write-Host "❌ 前端服务: 异常" -ForegroundColor Red
}
"@ | Out-File -FilePath "服务健康检查.ps1" -Encoding UTF8
```

### 2. 运行健康检查
```powershell
.\服务健康检查.ps1
```

## 🛠️ 常见问题解决

### 1. 端口冲突
```powershell
# 查看端口占用
netstat -ano | findstr :8080

# 结束占用进程
taskkill /PID <进程ID> /F
```

### 2. Maven依赖问题
```powershell
# 清理并重新编译
cd Memorin-rebuild\backend
mvn clean compile

# 或者单独清理某个服务
cd memorin-review-service
mvn clean compile
```

### 3. 前端依赖问题
```bash
# 清理并重新安装
cd Memorin-rebuild/frontend
rm -rf node_modules package-lock.json
npm install
```

### 4. Nacos连接问题
```powershell
# 检查Nacos是否正常启动
curl http://localhost:8848/nacos

# 如果无法访问，手动启动Nacos
cd nacos-server
.\bin\startup.cmd
```
